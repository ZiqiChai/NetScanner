cmake_minimum_required(VERSION 3.16)
project(NetScanner VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Network Widgets)
add_library(scan_core src/targets.cpp src/scanner.cpp)
target_include_directories(scan_core PUBLIC src)
target_link_libraries(scan_core PUBLIC Qt6::Core Qt6::Network)
add_library(scan_ui src/mainwindow.cpp)
target_link_libraries(scan_ui PUBLIC scan_core Qt6::Widgets)
add_executable(netscanner src/main.cpp)
target_link_libraries(netscanner PRIVATE scan_ui)
foreach(target scan_core scan_ui netscanner)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
if(BUILD_TESTING)
 find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
 foreach(test targets scanner ui nmap)
  add_executable(test_${test} tests/test_${test}.cpp)
  target_link_libraries(test_${test} PRIVATE scan_ui Qt6::Test)
  add_test(NAME ${test} COMMAND test_${test})
  set_tests_properties(${test} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
 endforeach()
endif()
install(TARGETS netscanner RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.ZiqiChai.NetScanner.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/io.github.ZiqiChai.NetScanner.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES packaging/io.github.ZiqiChai.NetScanner.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
